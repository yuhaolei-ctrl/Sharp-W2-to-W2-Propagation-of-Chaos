import SharpWasserstein.PeriodicSourceConvolutionPDEBasic

/-! Differentiation of actual scalar-weighted kernel integrals under arbitrary
random spatial labels. All differentiations use proved Bochner domination by
the integrable weight. -/
noncomputable section
open MeasureTheory Filter Set
open scoped Topology ContDiff BigOperators
namespace SharpWasserstein.PeriodicSourceConvolution
open NoiseAverage WeightedConvolution PeriodicIntegrationByParts PeriodicFourierTests

variable {Ω : Type} [MeasurableSpace Ω] {n : ℕ} (P : Measure Ω)
  (X : Ω → Coordinates n)

/-- A genuine translated scalar integral with an arbitrary integrable weight. -/
def scalarAverage (w : Ω → ℝ) (f : Coordinates n → ℝ) (x : Coordinates n) : ℝ :=
  ∫ q,w q*f (x-X q) ∂P

/-- The scalar weight acts by the actual bounded multiplication operator. -/
def scalarWeight : ℝ →L[ℝ] (ℝ →L[ℝ] ℝ) :=
  (ContinuousLinearMap.id ℝ ℝ).smulRight (ContinuousLinearMap.id ℝ ℝ)

theorem scalarAverage_eq_operatorAverage (w : Ω → ℝ) (f : Coordinates n → ℝ) :
    scalarAverage P X w f = operatorAverage P (fun q => -X q) (fun q => scalarWeight (w q)) f := by
  funext x
  unfold scalarAverage operatorAverage scalarWeight
  simp only [ContinuousLinearMap.smulRight_apply,ContinuousLinearMap.id_apply,smul_eq_mul,sub_eq_add_neg]

theorem scalarAverage_integrand_integrable (hX : Measurable X) {w : Ω → ℝ}
    (hw : Integrable w P) {f : Coordinates n → ℝ} (hf : Continuous f)
    {C : ℝ} (hC : ∀ x,‖f x‖ ≤ C) (x : Coordinates n) :
    Integrable (fun q => w q*f (x-X q)) P := by
  have h := operatorIntegrand_integrable P (fun q => -X q) hX.neg.stronglyMeasurable
    (scalarWeight.integrable_comp hw) hf hC x
  convert h using 1
  funext q
  simp only [scalarWeight,ContinuousLinearMap.smulRight_apply,ContinuousLinearMap.id_apply,
    smul_eq_mul,sub_eq_add_neg]

/-- The actual weighted convolution is C∞ for an arbitrary integrable scalar weight. -/
theorem scalarAverage_smooth (hX : Measurable X) {w : Ω → ℝ} (hw : Integrable w P)
    {f : Coordinates n → ℝ} (hf : ContDiff ℝ ∞ f) (hB : AllDerivativesBounded f) :
    ContDiff ℝ ∞ (scalarAverage P X w f) := by
  rw [scalarAverage_eq_operatorAverage]
  exact contDiff_infty_operatorAverage P (fun q => -X q) hX.neg.stronglyMeasurable
    (scalarWeight.integrable_comp hw) hf hB

/-- Exact first spatial derivative of the genuine weighted convolution. -/
theorem coordinatePartial_scalarAverage (hX : Measurable X) {w : Ω → ℝ} (hw : Integrable w P)
    {f : Coordinates n → ℝ} (hf : ContDiff ℝ ∞ f) (hB : AllDerivativesBounded f)
    (i : Fin n) (x : Coordinates n) :
    coordinatePartial (scalarAverage P X w f) i x = scalarAverage P X w (coordinatePartial f i) x := by
  obtain ⟨C,_,hC⟩ := hB.bounded
  obtain ⟨D,_,hD⟩ := hB.fderiv.bounded
  let W := fun q => scalarWeight (w q)
  have hW : Integrable W P := scalarWeight.integrable_comp hw
  let W' := fun q => ContinuousLinearMap.compL ℝ (Coordinates n) ℝ ℝ (W q)
  have hW' : Integrable W' P := (ContinuousLinearMap.compL ℝ (Coordinates n) ℝ ℝ).integrable_comp hW
  have hi := operatorIntegrand_integrable P (fun q => -X q) hX.neg.stronglyMeasurable hW'
    ((contDiff_infty_iff_fderiv.mp hf).2.continuous) hD x
  unfold coordinatePartial
  rw [scalarAverage_eq_operatorAverage,
    (hasFDerivAt_operatorAverage P (fun q => -X q) hX.neg.stronglyMeasurable hW
      (hf.of_le (ENat.natCast_le_of_coe_top_le_withTop le_rfl 1)) hC hD x).fderiv]
  rw [ContinuousLinearMap.integral_apply hi]
  unfold scalarAverage
  apply integral_congr_ae
  exact Eventually.of_forall fun q => by
    simp only [W,scalarWeight,ContinuousLinearMap.comp_apply,ContinuousLinearMap.smulRight_apply,
      ContinuousLinearMap.id_apply,smul_eq_mul,sub_eq_add_neg]

/-- Differentiating twice and summing retains the exact diffusion coefficient. -/
theorem laplacian_scalarAverage (hX : Measurable X) {w : Ω → ℝ} (hw : Integrable w P)
    {f : Coordinates n → ℝ} (hf : ContDiff ℝ ∞ f) (hB : AllDerivativesBounded f)
    (x : Coordinates n) :
    PeriodicIntegrationByParts.laplacian (scalarAverage P X w f) x =
      scalarAverage P X w (PeriodicIntegrationByParts.laplacian f) x := by
  have hb (i : Fin n) : AllDerivativesBounded (coordinatePartial f i) :=
    hB.fderiv.linear_comp (contDiff_infty_iff_fderiv.mp hf).2
      (ContinuousLinearMap.apply ℝ ℝ (Pi.single i 1))
  have he (i : Fin n) : coordinatePartial (scalarAverage P X w f) i =
      scalarAverage P X w (coordinatePartial f i) := funext (coordinatePartial_scalarAverage P X hX hw hf hB i)
  unfold PeriodicIntegrationByParts.laplacian
  simp_rw [he,coordinatePartial_scalarAverage P X hX hw (smooth_coordinatePartial hf _) (hb _)]
  unfold scalarAverage
  simp_rw [Finset.mul_sum]
  symm
  apply integral_finsetSum
  intro i _
  obtain ⟨C,_,hC⟩ := (hb i).fderiv.bounded
  apply scalarAverage_integrand_integrable P X hX hw
    (smooth_coordinatePartial (smooth_coordinatePartial hf i) i).continuous
  intro y
  exact (ContinuousLinearMap.le_opNorm _ _).trans
    (mul_le_mul_of_nonneg_right (hC y) (norm_nonneg _))

end SharpWasserstein.PeriodicSourceConvolution
